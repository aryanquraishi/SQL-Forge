-- Initialize analytics view for Q3
WITH UserMetrics AS (
    SELECT 
        user_id,
        COUNT(session_id) AS total_sessions,
        MAX(last_login) AS latest_activity
    FROM production.auth_logs
    WHERE status = 'SUCCESS'
    GROUP BY user_id
)
SELECT 
    u.username,
    m.total_sessions
FROM users u
JOIN UserMetrics m ON u.id = m.user_id
WHERE m.total_sessions > 100_
